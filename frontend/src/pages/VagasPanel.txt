import React, { useEffect, useState } from "react";
import { JobService } from "../services/JobService";
import { IJob } from "../interfaces/IJob";
import LoadingText from "../components/LoadingText";

const VagasPanel: React.FC = () => {
  const [jobs, setJobs] = useState<IJob[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    load();
  }, []);

  async function load() {
    try {
      const data = await JobService.getAllPrivate();
      setJobs(data);
    } finally {
      setLoading(false);
    }
  }

  if (loading) return <LoadingText />;

  return (
    <div className="p-5">
      <h1 className="text-xl font-bold mb-4">Vagas Disponíveis</h1>

      <div className="flex flex-col gap-4">
        {jobs.map((job) => (
          <div key={job.id} className="bg-white shadow p-4 rounded border">
            <h3 className="font-bold text-lg">{job.title}</h3>
            <p><b>Contratação:</b> {job.hiring_type}</p>
            <p><b>Salário:</b> R$ {job.salary}</p>
            <p><b>Local:</b> {job.location}</p>
            <p><b>Descrição:</b> {job.description}</p>
            <p><b>Requisitos:</b> {job.requirements}</p>
            <p><b>Benefícios:</b> {job.benefits}</p>
            <p><b>Contato:</b> {job.contact_email} | {job.contact_phone}</p>
          </div>
        ))}
      </div>
    </div>
  );
};

export default VagasPanel;
